# --------------------------------------------------
# PDF READER
# --------------------------------------------------

def extract_pdf_text(uploaded_file):

    text = ""

    try:
        with pdfplumber.open(uploaded_file) as pdf:

            for page in pdf.pages:

                page_text = page.extract_text()

                if page_text:
                    text += page_text + "\n"

        return text

    except Exception as e:

        st.error(f"PDF Reading Error: {e}")
        return ""


# --------------------------------------------------
# DYNAMIC PARAMETER EXTRACTION
# --------------------------------------------------

def extract_all_parameters(text):

    results = {}

    lines = text.splitlines()

    pattern = re.compile(
        r'([A-Za-z0-9\s()/%._-]+)\s*[:\-]\s*([0-9]+\.?[0-9]*)'
    )

    for line in lines:

        match = pattern.search(line)

        if match:

            parameter = match.group(1).strip()

            try:
                value = float(match.group(2))
                results[parameter] = value
            except:
                pass

    return results


# --------------------------------------------------
# ABNORMAL DETECTION
# --------------------------------------------------

def detect_abnormalities(report_data):

    abnormalities = []

    for parameter, value in report_data.items():

        found = reference_df[
            reference_df["parameter"].str.lower()
            ==
            parameter.lower()
        ]

        if found.empty:
            continue

        min_val = float(found.iloc[0]["min"])
        max_val = float(found.iloc[0]["max"])
        specialist = found.iloc[0]["specialist"]

        if value < min_val:

            abnormalities.append({
                "parameter": parameter,
                "value": value,
                "status": "LOW",
                "specialist": specialist
            })

        elif value > max_val:

            abnormalities.append({
                "parameter": parameter,
                "value": value,
                "status": "HIGH",
                "specialist": specialist
            })

    return abnormalities


# --------------------------------------------------
# SPECIALIST RECOMMENDATION
# --------------------------------------------------

def recommend_specialists(abnormalities):

    specialists = set()

    for item in abnormalities:
        specialists.add(item["specialist"])

    return sorted(list(specialists))


# --------------------------------------------------
# HEALTH SUMMARY
# --------------------------------------------------

def generate_summary(abnormalities):

    summary = []

    for item in abnormalities:

        summary.append(
            f"{item['parameter']} is {item['status']} "
            f"(Value: {item['value']})"
        )

    return summary


# --------------------------------------------------
# LIFESTYLE RECOMMENDATIONS
# --------------------------------------------------

def lifestyle_recommendations(abnormalities):

    recommendations = []

    for item in abnormalities:

        parameter = item["parameter"].lower()

        if "hba1c" in parameter:

            recommendations.append(
                "Reduce sugar intake and walk 30 minutes daily."
            )

        elif "ldl" in parameter:

            recommendations.append(
                "Avoid oily food and increase cardio exercise."
            )

        elif "vitamind" in parameter.replace(" ", ""):

            recommendations.append(
                "Increase sunlight exposure and Vitamin D foods."
            )

        elif "creatinine" in parameter:

            recommendations.append(
                "Stay hydrated and monitor kidney function."
            )

        elif "tsh" in parameter:

            recommendations.append(
                "Schedule a thyroid evaluation."
            )

        elif "hemoglobin" in parameter:

            recommendations.append(
                "Increase iron-rich foods."
            )

    if len(recommendations) == 0:

        recommendations.append(
            "Maintain a healthy diet and annual health checks."
        )

    return list(set(recommendations))


# --------------------------------------------------
# HEALTH SCORE
# --------------------------------------------------

def calculate_score(abnormalities):

    score = 100

    score -= (len(abnormalities) * 10)

    if score < 0:
        score = 0

    return score
